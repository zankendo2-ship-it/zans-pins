"""ZANS knowledge pack: everything the project knows, in one folder every Claude chat can read.

    python scripts/knowledge_pack.py   ->  reports/knowledge_pack/
        ZANS_product_data_pack.md   master summary (verified records, rules in force, methods index, research index)
        project_memory.md           copy of CLAUDE.md (full history of decisions)
        records/*.md                live trial records (copied as they stand)
        research/*                  every research write-up (md/txt) + helper reports
        methods/<course>_<CODE>.md  every course method in use
        rules/RULES.md, research_queue.md
        pack.py                     this script
        index.html                  the page (links to every file)

Published as one Artifact; other chats read it with Artifact action "read" + path. Rebuild + republish weekly
(Sunday evening review) and after any method change.
"""
import datetime as dt
import glob
import html
import json
import os
import re
import shutil
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "reports" / "knowledge_pack"


def read(p):
    try:
        return Path(p).read_text(encoding="utf-8", errors="replace")
    except OSError:
        return ""


def first_lines(text, n=3):
    out = []
    for line in text.splitlines():
        s = line.strip().lstrip("#").strip()
        if s and not s.startswith("|---"):
            out.append(s)
        if len(out) >= n:
            break
    return out


def slip_record():
    rows, staked, pl = [], 0.0, 0.0
    for f in sorted(glob.glob(str(ROOT / "reports/predictions/2026-*/slip_review.json"))):
        d = json.loads(read(f) or "{}")
        s, r = float(d.get("staked") or 0), float(d.get("returned") or 0)
        staked += s
        pl += r - s
        rows.append((Path(f).parent.name, d.get("counted_slip", ""), s, r - s, pl))
    return rows, staked, pl


def headline(path, pattern=r"\*\*.*\*\*"):
    m = re.findall(pattern, read(path))
    return m[-1].strip("*") if m else ""


def methods_index():
    rows = []
    for f in sorted(glob.glob(str(ROOT / "methods/*/*.md"))):
        p = Path(f)
        if p.parent.name in ("inbox",) or p.name.lower() == "readme.md":
            continue
        text = read(f)
        ver = re.search(r"\bv(\d+(?:\.\d+)+)", text)
        src = p.parent / "source"
        owner = src.is_dir() and any(src.iterdir())
        rows.append((p.parent.name, p.stem, "v" + ver.group(1) if ver else "?", "owner's method" if owner else "built from history"))
    return rows


def build():
    if OUT.exists():
        shutil.rmtree(OUT)
    for sub in ("records", "research", "methods", "rules"):
        (OUT / sub).mkdir(parents=True)

    shutil.copy(ROOT / "CLAUDE.md", OUT / "project_memory.md")
    shutil.copy(__file__, OUT / "pack.py.txt")
    for f in ("rules/RULES.md", "config/research_queue.md"):
        if (ROOT / f).exists():
            shutil.copy(ROOT / f, OUT / "rules" / Path(f).name)
    records = sorted(glob.glob(str(ROOT / "reports/predictions/*_record.md"))) + [
        str(ROOT / "reports/predictions/leak_finder.md"), str(ROOT / "reports/customer/record.md"),
        str(ROOT / "reports/angles/angles_record.md")]
    for f in records:
        if os.path.exists(f):
            shutil.copy(f, OUT / "records" / Path(f).name)
    research = sorted(glob.glob(str(ROOT / "reports/research/*.md")) + glob.glob(str(ROOT / "reports/research/*.txt"))
                      + glob.glob(str(ROOT / "reports/research/helpers/*.md")))
    for f in research:
        shutil.copy(f, OUT / "research" / Path(f).name)
    mrows = methods_index()
    for course, code, *_ in mrows:
        shutil.copy(ROOT / "methods" / course / f"{code}.md", OUT / "methods" / f"{course}_{code}.md")

    rows, staked, pl = slip_record()
    weights = json.loads(read(ROOT / "config/predict_weights.json") or "{}")
    feats = weights.get("features") or []
    today = dt.date.today().isoformat()

    md = [f"# ZANS Racing - product data pack ({today})", "",
          "Everything the ZANS project knows, in one file. Every figure below is copied from the project's own records "
          "(locked packs, counted bet slips, settled results). Tests on history are labelled TEST. UK and Irish racing only.",
          "", "Files in this pack: project_memory.md (every decision since 23 Sep), records/ (live trial records), "
          "research/ (every study), methods/ (every course method), rules/ (pre-registered tests + research queue).", "",
          "## 1. The bet slip - verified profit and loss (one counted slip per day, £10 a bet)", "",
          f"**Total: {len(rows)} days, staked £{staked:,.2f}, profit/loss {'+' if pl >= 0 else '-'}£{abs(pl):,.2f}.**", "",
          "| Date | Slip counted | Staked | Day P/L | Running |", "|---|---|---|---|---|"]
    md += [f"| {d} | {c} | £{s:.2f} | {p:+.2f} | {r:+.2f} |" for d, c, s, p, r in rows]
    md += ["", "## 2. Live records (as they stand in records/)", ""]
    for f in records:
        if os.path.exists(f):
            h = headline(f) or " / ".join(first_lines(read(f), 2))
            md.append(f"- **{Path(f).stem}**: {h}")
    md += ["", "## 3. How the picks are made (model v10)", "",
           f"- Win pick = conditional-logit model on {len(feats)} evidence features plus the market price, flat and jumps "
           "fitted separately, then each course's method overlay (rule factors tested on history in two periods).",
           "- Confidence 0-10 is a calibrated win chance. BET label = confidence 8+ AND the evidence-only model agrees.",
           "- Features: " + ", ".join(feats), "",
           "### Bet slip rules in force",
           "- Win singles + doubles/trebles/Lucky 15/63 from 7/10+ win picks (any price from 1.00, trial since 20 Oct).",
           "- Kept off every bet: 90+ days off; maiden/novice/beginners/bumper jumps; drifted 15%+; latest run missing or "
           "abroad (data gap); never run at the trip (and on going, jumps); Southwell AW handicaps; Fairyhouse jumps "
           "non-handicaps; heavy ground flat picks never on soft/heavy = single only (trial).",
           "- Each-way singles and forecasts OFF; high-odds each-way Lucky 15 (12.00+, Sky Bet) ON; spread multiples.",
           "- BET NOW emails: early value horses 15.00+ where Sky Bet pays EXTRA places.", "",
           "## 4. Course methods in use", "", "| Course | Code | Version | Source |", "|---|---|---|---|"]
    md += [f"| {c} | {k} | {v} | {s} |" for c, k, v, s in mrows]
    md += ["", "## 5. Research - every study (full text in research/)", ""]
    for f in research:
        lines = first_lines(read(f), 3)
        md.append(f"- **{Path(f).name}**: " + " - ".join(lines)[:400])
    md += ["", "## 6. Data sources", "",
           "- Results 2015-2026: Kaggle Racing Post extract (to May 2026) + our own Racing Post scrape (data-results) "
           "+ Sporting Life results; UK/IRE only.",
           "- Prices: Sporting Life feed (Betfair Sportsbook, Paddy Power, Sky Bet with places) at 5 checks a day.",
           "- Racecards: Racing Post (rpscrape) cross-checked with The Racing API.",
           "- Going, stewards' and vet reports: BHA (GB) and IHRB (Ireland). Weather: Open-Meteo.",
           "- Course facts: course_kb artifact https://claude.ai/artifact/AP2pBbB9RbymKEmcfn2Kyt",
           "- Live customer record: https://claude.ai/artifact/7Mq8DGaaQwFpZDHMXMb7hR", ""]
    (OUT / "ZANS_product_data_pack.md").write_text("\n".join(md), encoding="utf-8")

    files = sorted(p.relative_to(OUT).as_posix() for p in OUT.rglob("*") if p.is_file())
    write_index(files, rows, staked, pl, mrows, research, today)
    return files

                  for d, _, s, p, r in last)
    page = f"""<title>ZANS Knowledge Pack</title>

if __name__ == "__main__":
    fs = build()
    print(f"{len(fs)} files -> {OUT}")
